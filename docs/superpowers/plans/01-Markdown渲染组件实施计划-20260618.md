// packages/markdown/src/utils/extractCompleteMarkdown.ts
/**
 * 从流式 buffer 中提取「安全可渲染」的 Markdown：
 * 未闭合的代码围栏 ``` 或块级公式 $$ 会被截断到最后一个安全边界，
 * 避免未闭合块破坏后续渲染。行内 $ 不处理（单字符无法判断配对）。
 */
export function extractCompleteMarkdown(buffer: string): string {
  if (!buffer) return ''

  let safe = buffer

  // 代码围栏 ```：奇数个说明最后一个未闭合
  const fenceCount = (safe.match(/```/g) ?? []).length
  if (fenceCount % 2 === 1) {
    safe = safe.slice(0, safe.lastIndexOf('```'))
  }

  // 块级公式 $$：奇数个说明最后一个未闭合
  const dollarCount = (safe.match(/\$\$/g) ?? []).length
  if (dollarCount % 2 === 1) {
    safe = safe.slice(0, safe.lastIndexOf('$$'))
  }

  return safe
}
```

// packages/markdown/src/composables/useStreamingMarkdown.ts
import { shallowRef } from 'vue'
import DOMPurify from 'dompurify'
import { useMarkdownIt } from './useMarkdownIt'
import { extractCompleteMarkdown } from '../utils/extractCompleteMarkdown'

const ALLOWED_TAGS = DOMPurify.sanitize('', { RETURN_CONFIG: true })
  ? undefined
  : undefined // 使用默认白名单 + 保留 md 产物；mermaid svg 后续单独处理

/**
 * 流式 markdown 渲染：容错截断 + requestAnimationFrame 节流 + DOMPurify 消毒。
 * 返回 shallowRef 避免深度响应式开销。
 */
export function useStreamingMarkdown() {
  const md = useMarkdownIt()
  const html = shallowRef<string>('')

  let rafId = 0
  let pending = ''

  const renderNow = () => {
    const safe = extractCompleteMarkdown(pending)
    const raw = md.render(safe)
    html.value = DOMPurify.sanitize(raw, {
      // mermaid 的 svg 在 MermaidBlock 内部单独沙箱渲染，此处允许 svg 作为占位
      ADD_TAGS: ['svg', 'path', 'foreignObject'],
      ADD_ATTR: ['viewBox', 'd', 'fill', 'stroke'],
    })
  }

  const setContent = (content: string) => {
    pending = content
    if (rafId) cancelAnimationFrame(rafId)
    rafId = requestAnimationFrame(renderNow)
  }

  const flush = () => {
    if (rafId) cancelAnimationFrame(rafId)
    rafId = 0
    renderNow()
  }

  return { html, setContent, flush }
}
```

> 说明：`ALLOWED_TAGS` 那行无意义占位请删除——DOMPurify 默认白名单已足够，下方 sanitize 配置显式补 svg/path（为 Phase 4 Mermaid SVG 占位预留）。最终代码即：

```ts
// packages/markdown/src/composables/useStreamingMarkdown.ts（最终版）
import { shallowRef } from 'vue'
import DOMPurify from 'dompurify'
import { useMarkdownIt } from './useMarkdownIt'
import { extractCompleteMarkdown } from '../utils/extractCompleteMarkdown'

export function useStreamingMarkdown() {
  const md = useMarkdownIt()
  const html = shallowRef<string>('')

  let rafId = 0
  let pending = ''

  const renderNow = () => {
    const safe = extractCompleteMarkdown(pending)
    const raw = md.render(safe)
    html.value = DOMPurify.sanitize(raw, {
      ADD_TAGS: ['svg', 'path', 'foreignObject'],
      ADD_ATTR: ['viewBox', 'd', 'fill', 'stroke', 'xmlns'],
    })
  }

  const setContent = (content: string) => {
    pending = content
    if (rafId) cancelAnimationFrame(rafId)
    rafId = requestAnimationFrame(renderNow)
  }

  const flush = () => {
    if (rafId) cancelAnimationFrame(rafId)
    rafId = 0
    renderNow()
  }

  return { html, setContent, flush }
}
```

import {
  createHighlighterCore,
  type HighlighterCore,
} from 'shiki/core'
// @ts-expect-error 按需 grammar（shiki 细粒度 bundle）
import { createOnigurumaEngine } from 'shiki/engine/oniguruma'

// 按需语言（动态 import，仅打包实际使用的 grammar）
const langs = [
  import('shiki/langs/javascript.mjs'),
  import('shiki/langs/typescript.mjs'),
  import('shiki/langs/vue.mjs'),
  import('shiki/langs/jsx.mjs'),
  import('shiki/langs/tsx.mjs'),
  import('shiki/langs/json.mjs'),
  import('shiki/langs/bash.mjs'),
  import('shiki/langs/shell.mjs'),
  import('shiki/langs/html.mjs'),
  import('shiki/langs/css.mjs'),
  import('shiki/langs/python.mjs'),
  import('shiki/langs/go.mjs'),
  import('shiki/langs/rust.mjs'),
  import('shiki/langs/markdown.mjs'),
  import('shiki/langs/yaml.mjs'),
  import('shiki/langs/sql.mjs'),
].map((m) => m.then((mod) => mod.default))

const themes = [
  import('shiki/themes/github-light.mjs').then((m) => m.default),
  import('shiki/themes/github-dark.mjs').then((m) => m.default),
]

let promise: Promise<HighlighterCore> | null = null

/**
 * Shiki highlighter 单例（lazy async）。
 * Dual Themes：github-light / github-dark 同时加载，配合暗色模式 CSS 切换。
 */
export function getHighlighter(): Promise<HighlighterCore> {
  if (promise) return promise
  promise = createHighlighterCore({
    themes,
    langs,
    engine: createOnigurumaEngine(import('shiki/wasm')),
  })
  return promise
}
```

> 说明：若 `shiki/core` 细粒度 API 在当前 shiki 版本不可用，回退为 `createHighlighter({ langs: [...], themes: [...] })`（从 `shiki` 直接导入，全量）。两种都可行，优先细粒度以控体积。

// packages/markdown/src/renderers/tokensToHtml.ts
import type { ThemedToken } from '@shikijs/core'

function esc(s: string): string {
  return s
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
}

/**
 * ThemedToken[] → 高亮 HTML 字符串。
 * Dual Themes：用 CSS 变量 --shiki-light / --shiki-dark，配合暗色模式切换。
 */
export function tokensToHtml(tokens: ThemedToken[]): string {
  return tokens
    .map((t) => {
      const text = esc(t.content)
      if (t.htmlAttrs?.style) {
        return `<span style="${t.htmlAttrs.style}">${text}</span>`
      }
      // dual themes：若 token 带 light/dark 变体（htmlAttrs.style 由 shiki dual 注入）
      if (t.color) {
        return `<span style="color:${t.color}">${text}</span>`
      }
      return `<span>${text}</span>`
    })
    .join('')
}
```

// packages/markdown/src/composables/useShikiTokenizer.ts
import { ShikiStreamTokenizer } from '@shikijs/stream'
import type { ThemedToken } from '@shikijs/core'
import { getHighlighter } from './useHighlighter'
import { tokensToHtml } from '../renderers/tokensToHtml'

const KNOWN_LANGS = new Set([
  'js', 'javascript', 'ts', 'typescript', 'vue', 'jsx', 'tsx', 'json',
  'bash', 'sh', 'shell', 'html', 'css', 'python', 'py', 'go', 'rust',
  'markdown', 'md', 'yaml', 'yml', 'sql',
])

/** 把已有 token 数组转 HTML（用于 stable 缓存重渲） */
export function renderTokens(tokens: ThemedToken[]): string {
  return tokensToHtml(tokens)
}

/** 非流式：一次性渲染完整代码块 */
export function renderCode(code: string, lang: string): string {
  // sync 版需 highlighter 已就绪；异步包装见 renderCodeAsync
  return code
    .split('\n')
    .map((line) => esc(line))
    .join('\n')
}

function esc(s: string): string {
  return s.replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;')
}

/**
 * 流式渲染单个代码块：内部维护 ShikiStreamTokenizer，
 * grammar state 保证多行着色连续，stable+unstable 合并输出 HTML。
 */
export async function renderCodeStreaming(
  code: string,
  lang: string,
  isStreaming: boolean,
): Promise<string> {
  const highlighter = await getHighlighter()
  const safeLang = KNOWN_LANGS.has(lang) ? normalizeLang(lang) : 'text'

  if (safeLang === 'text' || !KNOWN_LANGS.has(lang)) {
    // 未知语言：纯文本
    return esc(code)
  }

  const tokenizer = new ShikiStreamTokenizer({
    highlighter,
    lang: safeLang,
    theme: 'github-light', // dual 模式可改为 themes: { light, dark }
  } as any)

  // 一次性 enqueue 整块（外层 markdown-it 已按闭合块切分）
  const { stable, unstable } = await tokenizer.enqueue(code)
  let tokens = stable
  if (isStreaming) {
    tokens = [...stable, ...unstable] // 流式中最后一行也显示（unstable）
  } else {
    const closed = tokenizer.close()
    tokens = [...stable, ...closed.stable]
  }

  return tokensToHtml(tokens)
}

function normalizeLang(lang: string): string {
  const map: Record<string, string> = {
    js: 'javascript', ts: 'typescript', py: 'python',
    sh: 'bash', shell: 'bash', md: 'markdown', yml: 'yaml',
  }
  return map[lang] ?? lang
}
```

> 策略：`useStreamingMarkdown` 产出 HTML 后，DOM 中存在 `<pre><code class="language-x">`、`<div data-mermaid>`。MarkdownRenderer 在 `v-html` 后用 `watchEffect + nextTick` 扫描这些节点，用 Vue 的 `createApp`/动态组件替换为 CodeBlock/MermaidBlock。

- [ ] **Step 1: 增强组件**

```vue
<!-- packages/markdown/src/MarkdownRenderer.vue（Phase 5 版） -->
<script setup lang="ts">
import { nextTick, onBeforeUnmount, ref, watchEffect } from 'vue'
import { createApp, type App } from 'vue'
import CodeBlock from './CodeBlock.vue'
import MermaidBlock from './MermaidBlock.vue'
import { useStreamingMarkdown } from './composables/useStreamingMarkdown'

const props = defineProps<{ content: string; streaming?: boolean }>()
const { html, setContent, flush } = useStreamingMarkdown()
const root = ref<HTMLElement | null>(null)
const mountedApps: App[] = []

async function mountSubComponents() {
  mountedApps.forEach((a) => a.unmount())
  mountedApps.length = 0
  await nextTick()
  const el = root.value
  if (!el) return

  // 代码块：把 <pre><code class="language-x"> 替换为 CodeBlock
  el.querySelectorAll('pre > code[class*="language-"]').forEach((codeEl) => {
    const pre = codeEl.parentElement!
    const lang = /language-(\w+)/.exec(codeEl.className)?.[1] ?? 'text'
    const code = codeEl.textContent ?? ''
    const host = document.createElement('div')
    pre.replaceWith(host)
    const app = createApp(CodeBlock, { code, language: lang, streaming: props.streaming })
    app.mount(host)
    mountedApps.push(app)
  })

  // mermaid 占位 → MermaidBlock
  el.querySelectorAll('[data-mermaid]').forEach((ph) => {
    const code = decodeURIComponent(ph.getAttribute('data-mermaid') ?? '')
    const app = createApp(MermaidBlock, { code })
    app.mount(ph as HTMLElement)
    mountedApps.push(app)
  })
}

watchEffect(() => setContent(props.content))
watchEffect(() => {
  if (!props.streaming) flush()
})
watchEffect(() => {
  if (html.value) mountSubComponents()
})

onBeforeUnmount(() => mountedApps.forEach((a) => a.unmount()))
</script>

<template>
  <div ref="root" class="ai-chat-markdown" v-html="html" />
</template>
```
